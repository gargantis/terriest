// Verification (not part of the spike): under the floor out, how does
// tunnel.test.ts's terrier 0 get to the far side of the ball? Same scene as
// the test. Run from the spike's folder: node <this> --out <json>
import { writeFileSync } from 'node:fs';
import { join, resolve } from 'node:path';
import { pathToFileURL } from 'node:url';

const ROOT = process.cwd();
const imp = (p) => import(pathToFileURL(join(ROOT, p)).href);
const { place, placeBall } = await imp('src/sim/scripts.ts');
const { run, scenario } = await imp('src/sim/testing.ts');
const { M0_SOLVER, parseSolver } = await imp('src/solver-settings.ts');
const args = process.argv.slice(2);
const out = resolve(args[args.indexOf('--out') + 1]);
const rows = [];
for (const spec of ['m0', 'nofloor']) {
  const sc = scenario(2, { ...M0_SOLVER, ...parseSolver(spec), drive: false, friction: false, gravity: 9.8 }, (s) => {
    placeBall(s, 0, 0);
    place(s, 0, -3, 0); s.terriers[0].vx = 25;
    place(s, 1, 3, 0.1); s.terriers[1].vx = -25;
    place(s, 2, -16, 4.1); s.terriers[2].vx = -25;
    place(s, 3, 5, 10); s.terriers[3].vy = 25;
  });
  let sub = 0;
  let prev = Math.sign(sc.s.terriers[0].x - sc.s.ball.x);
  let dMin = Infinity;
  const crossings = [];
  let zMax = 0;
  sc.sim.probe = (s) => {
    sub++;
    const t = s.terriers[0];
    const b = s.ball;
    dMin = Math.min(dMin, Math.hypot(t.x - b.x, t.y - b.y, t.z - b.z));
    zMax = Math.max(zMax, t.z);
    const side = Math.sign(t.x - b.x);
    if (side !== prev && Math.abs(t.y - b.y) < 0.5) crossings.push({ substep: sub, terrier: { x: +t.x.toFixed(3), y: +t.y.toFixed(3), z: +t.z.toFixed(3), vz: +t.vz.toFixed(3) }, ball: { x: +b.x.toFixed(3), y: +b.y.toFixed(3), z: +b.z.toFixed(3) }, heightAboveBallCentreM: +(t.z - b.z).toFixed(3) });
    prev = side;
  };
  run(sc, 60);
  rows.push({ spec, minCentreDistanceM: +dMin.toFixed(4), contactDistanceM: 1.05, terrier0MaxZ: +zMax.toFixed(3), crossings });
}
writeFileSync(out, `${JSON.stringify({ about: "Verification: how tunnel.test.ts's terrier 0 reaches the ball's far side with the floor out of the solve", utc: new Date().toISOString(), rows }, null, 1)}\n`);
console.log(JSON.stringify(rows));
